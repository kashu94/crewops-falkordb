# Cypher catalog

All Cypher queries used by CrewOps, generated by `scripts/cypher_catalog.py`.
The constant name is the query name shown in an answer's evidence.

## `crewops.agents.coordinator`

Shared state between agents: claims, handoffs and each user's inbox.

### REGISTER_AGENT

```cypher
MERGE (a:Agent {name: $name})
ON CREATE SET a.kind = $kind, a.started_ts = $now
SET a.seen_ts = $now
```

### CLAIM_NEXT

```cypher
MATCH (d:Disruption {status: 'REPORTED'})
WITH d ORDER BY d.reported_ts, d.id LIMIT 1
SET d.status = 'ANALYSING', d.claimed_by = $agent, d.claimed_ts = $now,
    d.attempts = coalesce(d.attempts, 0) + 1
WITH d
MERGE (a:Agent {name: $agent})
CREATE (a)-[:CLAIMED {ts: $now}]->(d)
RETURN d.id AS id, d.type AS type, d.attempts AS attempts
```

### RELEASE_FAILED

```cypher
MATCH (d:Disruption {id: $id})
SET d.status = CASE WHEN d.attempts >= $max THEN 'PARKED' ELSE 'REPORTED' END,
    d.last_error = $error
RETURN d.status AS status
```

### HANDOFF

```cypher
MATCH (d:Disruption {id: $id}), (a:Agent {name: $agent})
OPTIONAL MATCH (desk:Desk {id: $desk})
OPTIONAL MATCH (u:User {username: $user})
CREATE (h:Handoff {ts: $now, note: $note, kind: $kind})
CREATE (h)-[:FROM]->(a), (h)-[:ABOUT]->(d)
FOREACH (_ IN CASE WHEN desk IS NULL THEN [] ELSE [1] END | CREATE (h)-[:TO]->(desk))
FOREACH (_ IN CASE WHEN u IS NULL THEN [] ELSE [1] END | CREATE (h)-[:TO]->(u))
```

### ACK_HANDOFFS

```cypher
OPTIONAL MATCH (:User {username: $user})-[:STAFFS]->(desk:Desk)
WITH collect(desk.id) AS desks
MATCH (h:Handoff)-[:TO]->(t)
WHERE h.acked_ts IS NULL
  AND ((t:User AND t.username = $user) OR (t:Desk AND t.id IN desks))
SET h.acked_ts = $now, h.acked_by = $user
RETURN count(h) AS acked
```

### INBOX_CONTROLLER

```cypher
MATCH (u:User {username: $user})-[:STAFFS]->(desk:Desk)<-[:ASSIGNED_TO]-(d:Disruption)
WHERE d.status IN $open
OPTIONAL MATCH (h:Handoff)-[:ABOUT]->(d) WHERE h.acked_ts IS NULL
OPTIONAL MATCH (pl:Plan {status: 'REJECTED'})-[:FOR]->(d)
RETURN d.id AS disruption, d.type AS type, d.status AS status, d.summary AS summary,
       d.reported_ts AS reported_ts, count(DISTINCT h) AS new_handoffs,
       collect(DISTINCT pl.reject_reason) AS rejected
ORDER BY reported_ts
```

### INBOX_APPROVER

```cypher
MATCH (pl:Plan {status: 'SUBMITTED'})-[:FOR]->(d:Disruption)
MATCH (pl)-[:SUBMITTED_BY]->(s:User)
WHERE s.username <> $user
RETURN pl.id AS plan, d.id AS disruption, d.summary AS summary, pl.total_cost AS cost,
       s.name AS submitted_by, pl.submitted_ts AS submitted_ts
ORDER BY submitted_ts
```

### INBOX_REPORTER

```cypher
MATCH (d:Disruption)-[:REPORTED_BY]->(:User {username: $user})
OPTIONAL MATCH (d)-[:ASSIGNED_TO]->(desk:Desk)
RETURN d.id AS disruption, d.status AS status, d.summary AS summary, desk.id AS desk,
       d.reported_ts AS reported_ts
ORDER BY reported_ts DESC LIMIT 10
```

### INBOX_CREW

```cypher
MATCH (:User {username: $user})-[:IS_CREW]->(c:Crew)
OPTIONAL MATCH (d:Disruption)-[n:NOTIFIES]->(c)
OPTIONAL MATCH (c)-[a:ASSIGNED]->(p:Pairing) WHERE a.via_plan IS NOT NULL
RETURN c.id AS crew_id,
       collect(DISTINCT CASE WHEN d IS NULL THEN NULL ELSE {disruption: d.id, summary: d.summary, reason: n.reason} END) AS notices,
       collect(DISTINCT CASE WHEN p IS NULL THEN NULL ELSE {pairing: p.id, plan: a.via_plan, role: a.role} END) AS new_trips
```

### AGENT_ACTIVITY

```cypher
MATCH (a:Agent)
OPTIONAL MATCH (a)-[c:CLAIMED]->(d:Disruption)
OPTIONAL MATCH (h:Handoff)-[:FROM]->(a)
RETURN a.name AS agent, a.kind AS kind, a.seen_ts AS seen_ts, count(DISTINCT d) AS claimed,
       count(DISTINCT h) AS handoffs
ORDER BY agent
```

### USER_ROLES

```cypher
MATCH (u:User {username: $user}) RETURN u.roles AS roles, u.name AS name
```

## `crewops.agents.tools`

Tool registry shared by the assistant and the MCP server.

### LIST_DISRUPTIONS

```cypher
MATCH (d:Disruption)
WHERE $status IS NULL OR d.status = $status
OPTIONAL MATCH (d)-[:ASSIGNED_TO]->(k:Desk)
RETURN d.id AS id, d.type AS type, d.status AS status, d.summary AS summary, k.id AS desk
ORDER BY id DESC LIMIT 25
```

### NOTIFIED

```cypher
MATCH (d:Disruption {id: $id})-[n:NOTIFIES]->(t)
RETURN labels(t)[0] AS kind, coalesce(t.name, t.id) AS who, t.title AS title, t.team AS team,
       n.reason AS reason, n.sent_ts IS NOT NULL AS sent
ORDER BY kind, who
```

### CREW_PROFILE

```cypher
MATCH (c:Crew {id: $id})-[:BASED_AT]->(b:Station)
MATCH (c)-[:EMPLOYED_BY]->(al:Airline)
OPTIONAL MATCH (c)-[:RATED_ON]->(t:AircraftType)
WITH c, b, al, collect(t.name) AS ratings
OPTIONAL MATCH (c)-[a:ASSIGNED]->(p:Pairing)-[:HAS_DAY]->(d:DutyDay)
WITH c, b, al, ratings, d, p, a ORDER BY d.date
WITH c, b, al, ratings, collect(CASE WHEN d IS NULL THEN NULL ELSE {date: d.date, pairing: p.id, role: a.role} END) AS duties
OPTIONAL MATCH (c)-[oc:ON_CALL]->(:Station)
WITH c, b, al, ratings, duties, collect(oc.date) AS on_call
OPTIONAL MATCH (dis:Disruption)-[g:GROUNDS]->(c) WHERE dis.status IN $open
RETURN c.id AS crew_id, c.name AS name, c.rank AS rank, c.status AS status, b.code AS base,
       al.name AS airline, ratings, c.seniority AS seniority, c.experience_years AS experience_years,
       c.command_qualified AS command_qualified, c.right_seat_qualified AS right_seat_qualified,
       duties, on_call, collect(DISTINCT dis.id) AS grounded_by
```

### CREW_BY_NAME

```cypher
CALL db.idx.fulltext.queryNodes('Crew', $q) YIELD node
RETURN node.id AS crew_id, node.name AS name, node.rank AS rank LIMIT 5
```

### RESERVES

```cypher
MATCH (c:Crew)-[oc:ON_CALL]->(s:Station)
WHERE oc.date = $date AND ($base IS NULL OR s.code = $base) AND ($rank IS NULL OR c.rank = $rank)
MATCH (c)-[:EMPLOYED_BY]->(al:Airline)
OPTIONAL MATCH (dis:Disruption)-[:GROUNDS {date: $date}]->(c) WHERE dis.status IN $open
OPTIONAL MATCH (c)-[a:ASSIGNED]->(:Pairing)-[:HAS_DAY]->(d:DutyDay {date: $date})
RETURN c.id AS crew_id, c.name AS name, c.rank AS rank, s.code AS base, al.code AS airline,
       oc.start_ts AS window_start, oc.end_ts AS window_end,
       count(dis) > 0 AS grounded, count(d) > 0 AS already_flying
ORDER BY rank, crew_id
```


## `crewops.algorithms`

Graph algorithms used as tools: PageRank and betweenness over the route network

### PAGERANK

```cypher
CALL algo.pageRank('Station', 'ROUTE') YIELD node, score
RETURN node.code AS station, score
```

### BETWEENNESS

```cypher
CALL algo.betweenness({nodeLabels: ['Station'], relationshipTypes: ['ROUTE'],
                       samplingSize: 1000, samplingSeed: 7}) YIELD node, score
RETURN node.code AS station, score
```

### STATION_TRAFFIC

```cypher
MATCH (s:Station)
OPTIONAL MATCH (f:Flight)-[:FROM|TO]->(s)
WITH s, count(f) AS movements, sum(f.booked) AS passengers
OPTIONAL MATCH (c:Crew)-[:BASED_AT]->(s)
RETURN s.code AS station, movements, passengers, count(c) AS crew_based
```

### SP_PATHS

```cypher
MATCH (a:Station {code: $origin}), (b:Station {code: $dest})
CALL algo.SPpaths({sourceNode: a, targetNode: b, relTypes: ['ROUTE'], weightProp: 'block_hours',
                   maxLen: $max_legs, pathCount: $paths})
YIELD path, pathWeight
RETURN [n IN nodes(path) | n.code] AS stations, pathWeight AS block_hours
ORDER BY block_hours
```

### WCC

```cypher
CALL algo.WCC({nodeLabels: ['Station'], relationshipTypes: ['ROUTE']}) YIELD node, componentId
RETURN componentId AS component, collect(node.code) AS stations
```

### KNOCK_ON

```cypher
MATCH (f:Flight)
WHERE f.date = $date AND f.status <> 'cancelled'
OPTIONAL MATCH (f)-[:NEXT_LEG*1..8]->(g:Flight)
WHERE g.date = f.date
WITH f, count(g) AS downstream, sum(g.booked) AS passengers
MATCH (f)-[:ON]->(ac:Aircraft)
RETURN f.id AS flight, ac.tail AS aircraft, downstream, passengers
ORDER BY downstream DESC, passengers DESC
LIMIT $limit
```

### WHO_KNOWS_STATION

```cypher
MATCH (c:Crew)-[:ASSIGNED]->(:Pairing)-[:HAS_DAY]->(:DutyDay)-[:LEG]->(f:Flight)-[:TO]->(:Station {code: $station})
RETURN c.id AS crew_id, c.name AS name, c.rank AS rank, count(f) AS legs
ORDER BY legs DESC, crew_id
LIMIT $limit
```

### WHO_KNOWS_TYPE

```cypher
MATCH (c:Crew)-[:RATED_ON]->(:AircraftType {name: $type})
OPTIONAL MATCH (c)-[:ASSIGNED]->(:Pairing)-[:HAS_DAY]->(d:DutyDay)
RETURN c.id AS crew_id, c.name AS name, c.rank AS rank, c.experience_years AS years,
       count(d) AS duty_days
ORDER BY years DESC, duty_days DESC
LIMIT $limit
```

## `crewops.disruptions`

Report and analyse disruptions.

### USER_QUERY

```cypher
MATCH (u:User {username: $username})
OPTIONAL MATCH (u)-[:IS_CREW]->(c:Crew)
OPTIONAL MATCH (u)-[:STAFFS]->(d:Desk)
RETURN u.username AS username, u.name AS name, u.roles AS roles, c.id AS crew_id, d.id AS desk
```

### NEXT_ID

```cypher
MERGE (k:Counter {name: $name})
ON CREATE SET k.n = 0
SET k.n = k.n + 1
RETURN k.n AS n
```

### CREATE_DISRUPTION

```cypher
MATCH (u:User {username: $reporter})
CREATE (d:Disruption {id: $id, type: $type, status: 'REPORTED', reported_ts: $now,
                      summary: $summary, details: $details})
CREATE (d)-[:REPORTED_BY]->(u)
RETURN d.id AS id
```

### CONCERNS_CREW

```cypher
MATCH (d:Disruption {id: $id})
UNWIND $crew AS cid
MATCH (c:Crew {id: cid})
CREATE (d)-[:CONCERNS]->(c)
WITH d, c
UNWIND $dates AS day
CREATE (d)-[:GROUNDS {date: day}]->(c)
```

### CONCERNS_NODE

```cypher
MATCH (d:Disruption {id: $id})
MATCH (n) WHERE (n:Flight AND n.id = $ref) OR (n:Station AND n.code = $ref)
             OR (n:Gate AND n.id = $ref) OR (n:Aircraft AND n.tail = $ref)
CREATE (d)-[:CONCERNS]->(n)
```

### LOG_EVENT

```cypher
MATCH (d:Disruption {id: $id})
OPTIONAL MATCH (d)-[:LOGGED]->(prev:Event)
```

```cypher
MATCH (d:Disruption {id: $id})-[:REPORTED_BY]->(u:User)
```

### CONTENTION_QUERY

## `crewops.explain`

Explanations read back from the graph: decision trail, why a crew member was not

### TRAIL_QUERY

```cypher
MATCH (d:Disruption {id: $id})-[:REPORTED_BY]->(r:User)
OPTIONAL MATCH (d)-[:ASSIGNED_TO]->(desk:Desk)<-[:STAFFS]-(ctl:User)
OPTIONAL MATCH (d)-[:LOGGED]->(e:Event)
WITH d, r, desk, ctl, e ORDER BY e.seq
WITH d, r, desk, ctl, collect(e{.seq, .kind, .ts, .actor, .note}) AS events
OPTIONAL MATCH (pl:Plan)-[:FOR]->(d)
OPTIONAL MATCH (pl)-[:SUBMITTED_BY]->(s:User)
OPTIONAL MATCH (pl)-[:APPROVED_BY]->(a:User)
OPTIONAL MATCH (pl)-[:REJECTED_BY]->(rj:User)
OPTIONAL MATCH (pl)-[:HAS]->(x:Assignment)-[:ASSIGNS]->(c:Crew)
WITH d, r, desk, ctl, events, pl, s, a, rj,
     collect(CASE WHEN x IS NULL THEN NULL ELSE {gap: x.gap, crew: c.id, name: c.name,
             action: x.action, cost: x.cost, strategy: x.strategy} END) AS assigned
RETURN d{.id, .type, .status, .summary, .reported_ts} AS disruption,
       r.name AS reported_by, desk.id AS desk, ctl.name AS controller, events,
       collect(CASE WHEN pl IS NULL THEN NULL ELSE {
         plan: pl.id, status: pl.status, fingerprint: pl.fingerprint, seal: pl.seal,
         total_cost: pl.total_cost, submitted_by: s.name, approved_by: a.name,
         rejected_by: rj.name, reject_reason: pl.reject_reason, assignments: assigned} END) AS plans
```

### GAP_SUMMARY_QUERY

```cypher
MATCH (d:Disruption {id: $id})-[:HAS_GAP]->(g:Gap)-[:VACATED_BY]->(v:Crew)
OPTIONAL MATCH (g)-[o:OPTION]->(c:Crew)
WITH g, v, o, c ORDER BY o.rank
WITH g, v, collect(CASE WHEN c IS NULL THEN NULL ELSE {rank: o.rank, crew: c.id, name: c.name,
     cost: o.cost, strategy: o.strategy} END)[0..3] AS top
OPTIONAL MATCH (g)-[:RULED_OUT]->(x:Exclusion)-[:BREACHES]->(rule:Rule)
WITH g, v, top, rule.id AS rule, count(x) AS n
WITH g, v, top, collect(CASE WHEN rule IS NULL THEN NULL ELSE {rule: rule, excluded: n} END) AS breaches
RETURN g.id AS gap, g.pairing AS pairing, g.role AS role, v.id AS vacated_by, top, breaches
ORDER BY gap
```

### WHY_NOT_EXCLUDED

```cypher
MATCH (d:Disruption {id: $id})-[:HAS_GAP]->(g:Gap)-[:RULED_OUT]->(x:Exclusion)-[:CANDIDATE]->(c:Crew {id: $crew})
OPTIONAL MATCH (x)-[:BREACHES]->(r:Rule)
RETURN g.id AS gap, g.pairing AS pairing, c.name AS name, x.stage AS stage, x.reasons AS reasons,
       collect({rule: r.id, text: r.text}) AS rules
```

### WHY_NOT_OPTION

```cypher
MATCH (d:Disruption {id: $id})-[:HAS_GAP]->(g:Gap)-[o:OPTION]->(c:Crew {id: $crew})
OPTIONAL MATCH (g)-[better:OPTION]->(b:Crew) WHERE better.rank < o.rank
WITH g, o, c, better, b ORDER BY better.rank
RETURN g.id AS gap, g.pairing AS pairing, c.name AS name, o.rank AS rank, o.cost AS cost,
       o.action AS action,
       collect(CASE WHEN b IS NULL THEN NULL ELSE {rank: better.rank, crew: b.id, name: b.name,
               cost: better.cost} END) AS ranked_above
```

### RULE_PRESSURE_QUERY

```cypher
MATCH (d:Disruption)-[:HAS_GAP]->(:Gap)-[:RULED_OUT]->(x:Exclusion)-[:BREACHES]->(r:Rule)
WHERE d.status IN $statuses
RETURN r.id AS rule, r.text AS text, count(DISTINCT x) AS exclusions, count(DISTINCT d) AS disruptions
ORDER BY exclusions DESC
```

## `crewops.impact`

What a disruption affects.

### CREW_BLAST_QUERY

```cypher
MATCH (c:Crew {id: $crew})-[a:ASSIGNED]->(p:Pairing)-[:HAS_DAY]->(d:DutyDay)
WHERE d.date >= $from AND d.date <= $to AND d.report_ts >= $now
WITH c, a, p, collect(d.date) AS dates, min(d.date) AS first
MATCH (p)-[:HAS_DAY]->(dd:DutyDay)-[l:LEG]->(f:Flight)
WITH c, a, p, dates, first, dd, l, f ORDER BY dd.date, l.seq
WITH c, a, p, dates, first, collect({id: f.id, date: dd.date, booked: f.booked}) AS legs
MATCH (p)-[:FLOWN_BY]->(ac:Aircraft)
RETURN p.id AS pairing, a.role AS role, dates, ac.tail AS aircraft,
       [x IN legs WHERE x.date IN dates | x.id] AS uncovered,
       [x IN legs WHERE x.date > first AND NOT x.date IN dates | x.id] AS at_risk,
       reduce(s = 0, x IN [y IN legs WHERE y.date IN dates] | s + x.booked) AS passengers
ORDER BY pairing
```

### ROTATION_QUERY

```cypher
MATCH (f:Flight {id: $flight})
MATCH path = (f)-[:NEXT_LEG*0..12]->(g:Flight)
RETURN g.id AS id, g.no AS no, g.date AS date, g.dep_ts AS dep_ts, g.arr_ts AS arr_ts,
       g.booked AS booked, length(path) AS hop
ORDER BY hop
```

### FIRST_LEG_QUERY

```cypher
MATCH (f:Flight)-[:ON]->(:Aircraft {tail: $aircraft})
WHERE f.date = $date
RETURN f.id AS id ORDER BY f.dep_ts LIMIT 1
```

### DUTY_DAYS_OF_FLIGHTS_QUERY

```cypher
MATCH (dd:DutyDay)-[l:LEG]->(f:Flight)
WHERE f.id IN $flights
WITH DISTINCT dd
MATCH (dd)-[l2:LEG]->(leg:Flight)
WITH dd, leg ORDER BY l2.seq
WITH dd, collect({id: leg.id, dep_ts: leg.dep_ts, arr_ts: leg.arr_ts, booked: leg.booked}) AS legs
OPTIONAL MATCH (c:Crew)-[a:ASSIGNED]->(:Pairing {id: dd.pairing})
RETURN dd.id AS id, dd.pairing AS pairing, dd.date AS date, dd.report_ts AS report_ts,
       dd.release_ts AS release_ts, legs, collect({crew: c.id, role: a.role}) AS crew
ORDER BY dd.report_ts
```

### STATION_CLOSURE_QUERY

```cypher
MATCH (s:Station {code: $station})
MATCH (f:Flight)-[r:FROM|TO]->(s)
WHERE (type(r) = 'FROM' AND f.dep_ts >= $start AND f.dep_ts < $end)
   OR (type(r) = 'TO' AND f.arr_ts >= $start AND f.arr_ts < $end)
RETURN f.id AS id, type(r) AS direction, f.dep_ts AS dep_ts, f.arr_ts AS arr_ts, f.booked AS booked
ORDER BY f.dep_ts
```

### GATE_CLOSURE_QUERY

```cypher
MATCH (g:Gate {id: $gate})-[:AT]->(s:Station)
MATCH (f:Flight)-[b:BOARDS_AT]->(g)
WHERE b.start_ts < $end AND b.end_ts > $start
WITH s, f, b
OPTIONAL MATCH (alt:Gate)-[:AT]->(s)
WHERE alt.id <> $gate AND NOT alt.id IN $closed

WHERE alt IS NULL OR clashes = 0

       collect(alt.id) AS free_gates
ORDER BY start_ts
```

```cypher
MATCH (s:Session {id: $session})
OPTIONAL MATCH (s)-[:HAS]->(t:Turn)
WITH s, count(t) AS n
CREATE (s)-[:HAS]->(nt:Turn {seq: n + 1, ts: $now, question: $question, answer: $answer,
                             intent: $intent, tools: $tools, session: $session})

WITH s, nt
OPTIONAL MATCH (s)-[:HAS]->(prev:Turn {seq: nt.seq - 1})

WITH nt
```

### COMMIT_PLAN

```cypher
MATCH (pl:Plan {id: $plan}) WHERE pl.status = 'SUBMITTED' AND pl.fingerprint = $fp
MATCH (pl)-[:FOR]->(d:Disruption), (u:User {username: $user})
SET pl.status = 'COMMITTED', pl.approved_ts = $now, pl.seal = $seal, d.status = 'RESOLVED'
CREATE (pl)-[:APPROVED_BY {ts: $now}]->(u)
WITH pl
OPTIONAL MATCH (f:Flight) WHERE f.id IN $cancel
WITH pl, collect(f) AS cancelled
FOREACH (f IN cancelled | SET f.status = 'cancelled')
WITH pl
UNWIND (CASE WHEN size($moves) = 0 THEN [null] ELSE $moves END) AS m
OPTIONAL MATCH (mf:Flight {id: m.flight})-[old_gate:BOARDS_AT]->(:Gate)
OPTIONAL MATCH (ng:Gate {id: m.to_gate})
WITH pl, mf, ng, old_gate, old_gate.start_ts AS gs, old_gate.end_ts AS ge
DELETE old_gate
FOREACH (_ IN CASE WHEN mf IS NULL OR ng IS NULL THEN [] ELSE [1] END |
  CREATE (mf)-[:BOARDS_AT {start_ts: gs, end_ts: ge, via_plan: pl.id}]->(ng))
WITH pl, count(*) AS moved
UNWIND (CASE WHEN size($swaps) = 0 THEN [null] ELSE $swaps END) AS s
OPTIONAL MATCH (p:Pairing {id: s.pairing})
OPTIONAL MATCH (c:Crew {id: s.crew})
OPTIONAL MATCH (:Crew {id: s.vacated})-[old:ASSIGNED]->(p)
DELETE old
FOREACH (_ IN CASE WHEN p IS NULL OR c IS NULL THEN [] ELSE [1] END |
  CREATE (c)-[:ASSIGNED {role: s.role, via_plan: pl.id}]->(p))
WITH pl, moved, count(*) AS swapped
RETURN pl.status AS status, moved, swapped
```
